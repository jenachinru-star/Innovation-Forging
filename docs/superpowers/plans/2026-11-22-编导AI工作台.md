# 编导 AI 工作台 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 构建一个从"零散素材/已上传剧本"到"可排演剧本 + 排演准备表 + 人物关系图"的编导 AI 工作台，具备可完整演示的核心流程。

**Architecture:** 单体 Node.js 应用（npm workspaces 管理 `client`/`server`/`shared` 三个工作区）。后端 Express 提供 `/api/script/*` 接口，代理调用 DeepSeek（文本）与 Seedream（生图，加分项），无数据库、无状态；前端 React + Vite + Tailwind，Vite 开发时用 proxy 转发 `/api`，生产由 Express 托管 `client/dist`。核心契约 `ScriptData` 由 `shared` 工作区统一定义，前后端共同 import。

**Tech Stack:** Node.js v24、Express 5、React 19、Vite 8、Tailwind CSS 4（`@tailwindcss/vite`）、ECharts 6、mammoth（docx）、multer 2（上传）、Vitest 5 + supertest + @testing-library/react（测试，测试内 mock 外部 AI）。

## Global Constraints

- 技术栈必须使用 Node.js。
- 纯 JavaScript（`.js`/`.jsx`），不使用 TypeScript。
- 所有外部 AI Key 只存后端环境变量；前端不得出现任何密钥；真实密钥仅放 `server/.env`（不提交 git），仓库只提交 `server/.env.example`（仅变量名、无真实值）。
- 后端无状态、不引入数据库；数据随响应返回前端持有。
- 运行时调用真实 AI；自动化测试一律 mock 外部 AI 调用，不发真实网络请求。
- DeepSeek 环境变量名：`DEEPSEEK_API_KEY`、`DEEPSEEK_BASE_URL`、`DEEPSEEK_MODEL`。Seedream 环境变量名：`SEEDREAM_API_KEY`、`SEEDREAM_BASE_URL`、`SEEDREAM_MODEL`。后端端口：`PORT`（默认 3001）。
- `ScriptData` 契约（字段固定，不得随任务改名）：

```js
{
  summary: string,
  scenes: [{ sceneNumber: number, title: string, location: string,
             characters: string[], props: string[], summary: string,
             estimatedMinutes: number }],
  characters: [{ name: string, description: string }],
  relationships: [{ from: string, to: string, relation: string, needsReview: boolean }]
}
```

```js
function toStr(v) {
  return typeof v === 'string' ? v : v == null ? '' : String(v);
}
function toStrArray(v) {
  if (Array.isArray(v)) return v.map(toStr).filter(Boolean);
  if (typeof v === 'string' && v.trim()) return [v.trim()];
  return [];
}
function toNum(v) {
  const n = Number(v);
  return Number.isFinite(n) ? n : 0;
}

export function createEmptyScriptData() {
  return { summary: '', scenes: [], characters: [], relationships: [] };
}

export function normalizeScene(raw) {
  const r = raw && typeof raw === 'object' ? raw : {};
  return {
    sceneNumber: toNum(r.sceneNumber),
    title: toStr(r.title),
    location: toStr(r.location),
    characters: toStrArray(r.characters),
    props: toStrArray(r.props),
    summary: toStr(r.summary),
    estimatedMinutes: toNum(r.estimatedMinutes),
  };
}

export function normalizeCharacter(raw) {
  const r = raw && typeof raw === 'object' ? raw : {};
  return { name: toStr(r.name), description: toStr(r.description) };
}

export function normalizeRelationship(raw) {
  const r = raw && typeof raw === 'object' ? raw : {};
  return {
    from: toStr(r.from),
    to: toStr(r.to),
    relation: toStr(r.relation),
    needsReview: Boolean(r.needsReview),
  };
}

export function normalizeScriptData(raw) {
  const r = raw && typeof raw === 'object' ? raw : {};
  return {
    summary: toStr(r.summary),
    scenes: Array.isArray(r.scenes) ? r.scenes.map(normalizeScene) : [],
    characters: Array.isArray(r.characters) ? r.characters.map(normalizeCharacter) : [],
    relationships: Array.isArray(r.relationships) ? r.relationships.map(normalizeRelationship) : [],
  };
}
```

```js
const MODEL = process.env.DEEPSEEK_MODEL || 'deepseek-flash';
const KEY = process.env.DEEPSEEK_API_KEY || '';

export async function chatComplete(messages, { jsonMode = false } = {}) {
  const body = { model: MODEL, messages, temperature: 0.7 };
  if (jsonMode) body.response_format = { type: 'json_object' };

  const res = await fetch(`${BASE}/chat/completions`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      Authorization: `Bearer ${KEY}`,
    },
    body: JSON.stringify(body),
  });

  if (!res.ok) {
    const t = await res.text().catch(() => '');
    throw new Error(`DeepSeek API ${res.status}: ${t.slice(0, 200)}`);
  }
  const data = await res.json();
  const content = data?.choices?.[0]?.message?.content;
  if (typeof content !== 'string') throw new Error('DeepSeek 返回内容为空');
  return content;
}
```

- [ ] **Step 4: 实现 `server/services/llm.js`（本任务先实现 withRetry、generateScript、safeParse 骨架，parseScript 在 Task 4 补齐）**

```js
import { normalizeScriptData } from '@director/shared';
import { chatComplete } from './deepseekClient.js';

export async function withRetry(fn, retries = 2) {
  let lastErr;
  for (let i = 0; i <= retries; i += 1) {
    try {
      return await fn(i);
    } catch (e) {
      lastErr = e;
    }
  }
  throw lastErr;
}

const GENERATE_SYSTEM = [
  '你是一位资深戏剧编剧，帮助编导把零散素材整理成规范剧本。',
  '规则：',
  '1) 如果用户素材信息严重不足、无法展开成一场戏，请只回复一行，以 "NEED_MORE_INFO:" 开头，后面用一句自然的话反问缺失的关键信息（如人物名字、关系、发生了什么），不要编造。',
  '2) 否则，直接输出规范剧本文本：先写分场大纲（每场的核心人物与事件），再逐场展开【场次标题/场景地点/出场人物/舞台提示/人物对白】。',
  '3) 全文只输出剧本内容本身，不要额外解释。',
].join('\n');

export async function generateScript(material, { chat = chatComplete } = {}) {
  const messages = [
    { role: 'system', content: GENERATE_SYSTEM },
    { role: 'user', content: String(material || '') },
  ];
  const content = await withRetry(() => chat(messages), 2);
  const trimmed = content.trim();

  if (trimmed.startsWith('NEED_MORE_INFO:')) {
    const question = trimmed.slice('NEED_MORE_INFO:'.length).trim();
    return { needMoreInfo: true, question };
  }
  return { needMoreInfo: false, scriptText: trimmed };
}

export function safeParseScriptData(text) {
  if (typeof text !== 'string') return null;
  try {
    return normalizeScriptData(JSON.parse(text));
  } catch {
    /* fallthrough */
  }
  const start = text.indexOf('{');
  const end = text.lastIndexOf('}');
  if (start !== -1 && end !== -1 && end > start) {
    try {
      return normalizeScriptData(JSON.parse(text.slice(start, end + 1)));
    } catch {
      /* fallthrough */
    }
  }
  return null;
}
```

```js
  '你是戏剧结构分析助手。请把用户提供的剧本解析为严格 JSON，且只输出 JSON。',
  'JSON 结构（字段名必须完全一致）：',

  '  "scenes": [{ "sceneNumber": 1, "title": "第一场", "location": "地点",',
  '              "characters": ["出场人物"], "props": ["道具"],',
  '              "summary": "本场剧情摘要", "estimatedMinutes": 5 }],',
  '  "characters": [{ "name": "人物名", "description": "人物简介" }],',
  '  "relationships": [{ "from": "人物A", "to": "人物B",',
  '                     "relation": "关系描述", "needsReview": false }]',
  '}',
  '要求：estimatedMinutes 为估算的演出分钟数（整数）。',
  'relationships 尽量覆盖主要人物关系；若对某条关系判断把握不足，把其 needsReview 设为 true。',
  '无法判断的关系不要编造；没有任何关系时返回空数组。',
].join('\n');

export async function parseScript(scriptText, { chat = chatComplete } = {}) {
  const messages = [
    { role: 'system', content: PARSE_SYSTEM },
    { role: 'user', content: String(scriptText || '') },
  ];

  let content;
  try {
    content = await withRetry(() => chat(messages, { jsonMode: true }), 1);
  } catch {
    content = await withRetry(() => chat(messages), 1);
  }

  const first = safeParseScriptData(content);
  if (first) return first;

  const strongMessages = [
    { role: 'system', content: `${PARSE_SYSTEM}\n只输出纯 JSON，不要任何解释文字，不要代码块标记。` },
    { role: 'user', content: String(scriptText || '') },
  ];
  content = await withRetry(() => chat(strongMessages, { jsonMode: true }), 1);
  const second = safeParseScriptData(content);
  if (!second) throw new Error('解析失败：AI 输出无法解析为结构化数据');
```

```js
import path from 'node:path';
import mammoth from 'mammoth';

export const ALLOWED_EXTENSIONS = ['.txt', '.md', '.docx'];

export async function extractText(buffer, originalname) {
  const ext = path.extname(originalname || '').toLowerCase();
  if (!ALLOWED_EXTENSIONS.includes(ext)) {
    throw new Error(`不支持的文件类型：${ext || '(无扩展名)'}`);
  }
  if (ext === '.docx') {
    const result = await mammoth.extractRawText({ buffer });
    return String(result.value || '').trim();
  }
  return Buffer.from(buffer).toString('utf8').trim();
}
```

```js
import { Router } from 'express';
import multer from 'multer';
import * as llm from '../services/llm.js';
import * as extractService from '../services/extract.js';

const upload = multer({
  storage: multer.memoryStorage(),
  limits: { fileSize: 5 * 1024 * 1024 },
});

export function createScriptRouter(deps = {}) {
  const generateScript = deps.generateScript || llm.generateScript;
  const parseScript = deps.parseScript || llm.parseScript;
  const extractText = deps.extractText || extractService.extractText;

  const router = Router();

  router.post('/generate', async (req, res) => {
    const material = typeof req.body?.material === 'string' ? req.body.material.trim() : '';
    if (!material) return res.status(400).json({ error: '请提供素材内容' });
    try {
      const result = await generateScript(material);
      if (result.needMoreInfo) {
        return res.json({ error: result.question, needMoreInfo: true });
      }
      return res.json({ scriptText: result.scriptText });
    } catch (e) {
      return res.status(502).json({ error: '生成失败，请重试', detail: String(e?.message || e) });
    }
  });

  router.post('/extract', upload.single('file'), async (req, res) => {
    if (!req.file) return res.status(400).json({ error: '未收到文件' });
    try {
      const scriptText = await extractText(req.file.buffer, req.file.originalname);
      if (!scriptText) return res.status(400).json({ error: '文件内容为空' });
      return res.json({ scriptText });
    } catch (e) {
      return res.status(400).json({ error: String(e?.message || e) });
    }
  });

  router.post('/parse', async (req, res) => {
    const scriptText = typeof req.body?.scriptText === 'string' ? req.body.scriptText.trim() : '';
    if (!scriptText) return res.status(400).json({ error: '请提供剧本内容' });
    try {
      const scriptData = await parseScript(scriptText);
      return res.json({ scriptData });
    } catch (e) {
      return res.status(502).json({ error: '解析失败，请重试', detail: String(e?.message || e) });
    }
  });

  return router;
}
```

```js
async function readJson(res) {
  return res.json().catch(() => ({}));

export async function generateScript(material) {
  const res = await fetch('/api/script/generate', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ material }),
  });
  const data = await readJson(res);
```

```jsx
import ConfirmStep from './components/ConfirmStep.jsx';
import ResultTabs from './components/ResultTabs.jsx';
import { generateScript, parseScript } from './api.js';

export default function App() {
  const [step, setStep] = useState('input'); // input | confirm | result
  const [material, setMaterial] = useState('');
  const [scriptText, setScriptText] = useState('');
  const [entryMode, setEntryMode] = useState('generate'); // generate | upload
  const [scriptData, setScriptData] = useState(null);
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState('');

  function handleReady({ scriptText: text, material: mat, mode }) {
    setScriptText(text);
    setMaterial(mat || '');
    setEntryMode(mode);
    setError('');
    setStep('confirm');
  }

  async function handleRegenerate() {
    setBusy(true);
    setError('');
    try {
      const data = await generateScript(material);
      if (data.needMoreInfo) throw new Error(data.error);
      setScriptText(data.scriptText);
    } catch (e) {
      setError(String(e.message || e));
    } finally {
      setBusy(false);
    }
  }

  async function handleConfirm() {
    setBusy(true);
    setError('');
    try {
      const data = await parseScript(scriptText);
      setScriptData(data.scriptData);
      setStep('result');
    } catch (e) {
      setError(String(e.message || e));
    } finally {
      setBusy(false);
    }
  }

  return (
    <div className="min-h-screen bg-[#141010] text-[#f5ece0]">
      {step === 'input' && (
        <InputStep onReady={handleReady} busy={busy} setBusy={setBusy} error={error} setError={setError} />

      {step === 'confirm' && (
        <ConfirmStep
          scriptText={scriptText}
          onChangeScriptText={setScriptText}
          canRegenerate={entryMode === 'generate'}
          onRegenerate={handleRegenerate}
          onConfirm={handleConfirm}
          busy={busy}
          error={error}
        />
      )}
      {step === 'result' && (
        <ResultTabs scriptData={scriptData} onBackToEdit={() => setStep('confirm')} />
      )}
    </div>
  );
}
```

```jsx
import { useEffect, useRef } from 'react';

function buildOption(scriptData) {
  const names = Array.from(
    new Set([
      ...(scriptData.characters || []).map((c) => c.name),
      ...(scriptData.relationships || []).flatMap((r) => [r.from, r.to]),
    ])
  ).filter(Boolean);

  const nodes = names.map((n) => ({ name: n, symbolSize: 46 }));
  const links = (scriptData.relationships || []).map((r) => ({
    source: r.from,
    target: r.to,
    value: r.relation,
    label: { show: true, formatter: r.relation },
    lineStyle: {
      color: r.needsReview ? '#f59e0b' : '#a8a29e',
      width: r.needsReview ? 3 : 1.5,
      type: r.needsReview ? 'dashed' : 'solid',
    },
  }));

  return {
    tooltip: {},
    series: [{
      type: 'graph',
      layout: 'force',
      roam: true,
      label: { show: true, position: 'right', color: '#f5ece0' },
      force: { repulsion: 220, edgeLength: 120 },
      data: nodes,
      links,
      lineStyle: { color: '#a8a29e' },
      itemStyle: { color: '#d4a373' },
    }],
  };
}

export default function RelationGraph({ scriptData }) {
  const ref = useRef(null);
  const relationships = scriptData?.relationships || [];

  useEffect(() => {
    if (!ref.current || relationships.length === 0) return undefined;
    let chart = null;
    let disposed = false;

    import('echarts').then((echarts) => {
      if (disposed || !ref.current) return;
      chart = echarts.init(ref.current);
      chart.setOption(buildOption(scriptData));
    });

    return () => {
      disposed = true;
      if (chart) chart.dispose();
    };
  }, [scriptData, relationships.length]);

  if (relationships.length === 0) {
    return <p className="text-sm text-stone-400">暂无足够的人物关系信息。</p>;
  }

  return (
    <div>
      <p className="mb-2 text-xs text-stone-500">
        <span className="text-amber-400">橙色虚线</span> 表示 AI 判断把握不足，建议核实。
      </p>
      <div ref={ref} className="h-[420px] w-full rounded-lg border border-stone-800 bg-stone-900/40" />
    </div>
  );
}
```

```js
const BASE = process.env.SEEDREAM_BASE_URL || 'https://ark.cn-beijing.volces.com/api/v3';
const MODEL = process.env.SEEDREAM_MODEL || '';
const KEY = process.env.SEEDREAM_API_KEY || '';

export const STYLE_PROMPT = '水彩插画风格，暖色调，半身肖像，简化背景，柔和线条，戏剧海报质感';

export async function generateCharacterImage(description, { fetchImpl = fetch } = {}) {
  const prompt = `人物肖像：${description}。${STYLE_PROMPT}，非写实，不要照片级真人脸。`;
  const res = await fetchImpl(`${BASE}/images/generations`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${KEY}` },
    body: JSON.stringify({ model: MODEL, prompt, size: '1024x1024', response_format: 'url' }),
  });
  if (!res.ok) {
    const t = await res.text().catch(() => '');
    throw new Error(`Seedream API ${res.status}: ${t.slice(0, 200)}`);
  }
  const data = await res.json();
  const url = data?.data?.[0]?.url || data?.data?.[0]?.b64_json;
  if (!url) throw new Error('Seedream 未返回图片');
  return url;
}
```

```js
export function diffScriptData(a, b) {
  const sa = a?.scenes || [];
  const sb = b?.scenes || [];
  const mapA = new Map(sa.map((s) => [s.title, s]));
  const mapB = new Map(sb.map((s) => [s.title, s]));

  const addedScenes = [...mapB.keys()].filter((t) => !mapA.has(t));
  const removedScenes = [...mapA.keys()].filter((t) => !mapB.has(t));
  const changedScenes = [];
  for (const [title, sa1] of mapA) {
    const sb1 = mapB.get(title);
    if (sb1 && sa1.summary !== sb1.summary) {
      changedScenes.push({ title, from: sa1.summary, to: sb1.summary });
    }
  }

  const ca = new Set((a?.characters || []).map((c) => c.name));
  const cb = new Set((b?.characters || []).map((c) => c.name));
  return {
    addedScenes,
    removedScenes,
    changedScenes,
    characterDelta: {
      added: [...cb].filter((n) => !ca.has(n)),
      removed: [...ca].filter((n) => !cb.has(n)),
    },
  };
}
```

```jsx
import { useState } from 'react';
import { extractUpload, parseScript } from '../api.js';
import { diffScriptData } from '@director/shared/diff.js';

export default function VersionCompare() {
  const [a, setA] = useState(null);
  const [b, setB] = useState(null);
  const [diff, setDiff] = useState(null);
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState('');

  async function load(file, setter) {
    setBusy(true);
    setError('');
    try {
      const { scriptText } = await extractUpload(file);
      const { scriptData } = await parseScript(scriptText);
      setter(scriptData);
    } catch (e) {
      setError(String(e.message || e));
    } finally {
      setBusy(false);
    }
  }

  function compare() {
    if (a && b) setDiff(diffScriptData(a, b));
  }

  return (
    <div className="space-y-4 text-sm">
      <div className="grid grid-cols-2 gap-4">
        <div className="rounded border border-stone-700 p-3">
          <p className="mb-2 text-stone-400">旧版本（上传）</p>
          <input type="file" accept=".txt,.md,.docx" onChange={(e) => e.target.files?.[0] && load(e.target.files[0], setA)} />
          <p className="mt-2 text-xs text-stone-500">{a ? `已解析 ${a.scenes.length} 场` : '未加载'}</p>
        </div>
        <div className="rounded border border-stone-700 p-3">
          <p className="mb-2 text-stone-400">新版本（上传）</p>
          <input type="file" accept=".txt,.md,.docx" onChange={(e) => e.target.files?.[0] && load(e.target.files[0], setB)} />
          <p className="mt-2 text-xs text-stone-500">{b ? `已解析 ${b.scenes.length} 场` : '未加载'}</p>
        </div>
      </div>
      <button onClick={compare} disabled={!a || !b || busy} className="rounded bg-amber-200 px-4 py-2 text-stone-900 disabled:opacity-50">对比</button>
      {error && <p className="rounded bg-red-900/40 p-3 text-red-200">{error}</p>}
      {diff && (
        <div className="space-y-2">
          <p>新增场次：{diff.addedScenes.join('、') || '无'}</p>
          <p>删除场次：{diff.removedScenes.join('、') || '无'}</p>
          <p>改动场次：{diff.changedScenes.map((c) => c.title).join('、') || '无'}</p>
          <p>新增人物：{diff.characterDelta.added.join('、') || '无'}；删除人物：{diff.characterDelta.removed.join('、') || '无'}</p>

}
```
